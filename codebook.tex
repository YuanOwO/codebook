\subsection{公式們}
\begin{gather*}
    \begin{aligned}
        \sum_{k=1}^{n} n = \frac{n (n+1)}{2}, \quad          &
        \sum_{k=1}^{n} n^2 = \frac{n (n+1) (2n+1)}{6}, \quad &
        \sum_{k=1}^{n} n^3 = \bigg[ \frac{n (n+1)}{2} \bigg]^2
    \end{aligned}  \\
    \begin{aligned}
        \sum_{k=1}^{n} n^4 & = n (1+n) (1+2n) (-1 + 3n + 3n^2) \,/\, 30 \\
        \sum_{k=1}^{n} n^5 & = n^2 (n+1)^2 (2n^2+2n-1) \,/\, 12
    \end{aligned}  \\
    \begin{aligned}
        \textrm{錯位排列} \qquad &
        DP_i = \begin{cases}
                   0 \lor 1,                          & i = 1 \lor 2    \\
                   (i - 1)(DP_{i-1} \times DP_{i-2}), & i \geq 3        \\
                   = i DP_{i-1} + (-1)^i              & \textrm{另一種關係式}
               \end{cases}
    \end{aligned}  \\
    \begin{aligned}
        P_{k}^{n}   & = \frac{n!}{k!(n-k)!},         &
        C_{k}^{n}   & = \frac{n!}{(n-k)!},           &
        H_{k}^{n}   & = C_{n - 1}^{n + k -1}           \\
        \pi         & = \arccos(-1),                 &
        e^{i\theta} & = \cos \theta + i \sin \theta, &
        pV          & = nTR
    \end{aligned}  \\
    \begin{aligned}[c]
        a^{\varphi(m)} & \equiv 1 (\mod m)                        \\
        a^b            & \equiv a^{b \mod \varphi(m)-1} (\mod m)  \\
        \varphi(m)     & = m - 1, \text{When } m \text{ is Prime}
    \end{aligned}
\end{gather*}

\subsection{Pick's Theory}
\begin{align*}
    A = i + \frac{b}{2} - 1, \quad &
    2A = 2i + b - 2, \quad         &
    i = A - \frac{b}{2} + 1
\end{align*}
其中 $A$ 為面積、$b$ 為圖形內部的點數量、$i$ 為圖形線上的點數量

\subsection{位數 DP}
\noindent
\texttt{dp[位數][狀態]，dp[pos][state]}: \\
定義為目前位數在前導狀態為 \texttt{state} 的時候的計數 \\
例如：求數字沒有出現 \texttt{66} 的數量 \texttt{l \~{} r} \\
$\rightarrow$ \texttt{dp[pos][1]} 可表示計算 \texttt{pos} 個位數在前導出現一個 \texttt{6} 的計數 \\
$\rightarrow$ \texttt{dp[3][1]} 則計算 \texttt{6XXX} \\
模板的 \texttt{pos} 是反過來的，但不影響 (只是用來 \texttt{dp} 記憶用) \\
\begin{tabular}{l l}
    \texttt{pos}:   & 目前位數                                         \\
    \texttt{state}: & 前導狀態                                         \\
    \texttt{lead}:  & 是否有前導 \texttt{0}                             \\
                    & (大部分題目不用但有些數字 e.g. \texttt{00146} 如果有影響時要考慮) \\
    \texttt{limit}: & 使否窮舉有被 \texttt{num} 限制                       \\
\end{tabular}
\inputcpp{DP/digit.cpp}
